# M1 本地播放器 — 详细搭建步骤

> 目标：最小播放器，播一个 mp4（画面 + 声音 + 基本同步）。
> 分工：你写 C 内核（解码 + 同步 + 队列），我写 OC 胶水（Metal + AudioQueue + UI）。
> 参照：WorkLabs `WLMediaSource`（解码管线）、`WLAudioRenderer`（播放）、`WLNodeQueue`（队列）。

---

## 第三步：C 内核 — wl_common.h（公共类型）

你来写。这是所有模块共享的类型定义。

**文件**：`libwl/include/wl_common.h`

```c
#ifndef WL_COMMON_H
#define WL_COMMON_H

#include <stdint.h>
#include <stdbool.h>

// ─── 错误码 ───
#define WL_OK           0
#define WL_ERR_EOF     -1
#define WL_ERR_AGAIN   -2  // 需要更多数据
#define WL_ERR_NOMEM   -3
#define WL_ERR_IO      -4

// ─── 视频帧 ───
typedef struct {
    uint8_t *data;      // BGRA 像素数据（调用方负责 free）
    int width;
    int height;
    int linesize;       // 每行字节数（可能 > width*4，有 padding）
    double pts;         // 显示时间戳（秒）
} wl_video_frame;

// ─── 音频帧 ───
typedef struct {
    float *data;        // interleaved float PCM（调用方负责 free）
    int nb_samples;     // 采样数
    int channels;       // 声道数
    int sample_rate;    // 采样率
    double pts;         // 显示时间戳（秒）
} wl_audio_frame;

// ─── 释放帧内部数据 ───
static inline void wl_video_frame_free(wl_video_frame *f) {
    if (f && f->data) { free(f->data); f->data = NULL; }
}
static inline void wl_audio_frame_free(wl_audio_frame *f) {
    if (f && f->data) { free(f->data); f->data = NULL; }
}

#endif // WL_COMMON_H
```

**要点**：
- `wl_video_frame.data` 是 BGRA 格式（每像素 4 字节），可直接映射 Metal 纹理
- `wl_audio_frame.data` 是 interleaved float PCM（左右左右...），可直接给 AudioQueue
- `pts` 统一用秒（已经乘过 time_base）
- 帧数据由调用方 `free()`，结构体本身是栈上或嵌入的

## 第四步：C 内核 — wl_queue.h（线程安全队列）

你来写。参照 WorkLabs `WLNodeQueue`，但用纯 C 实现。

**文件**：`libwl/include/wl_queue.h`

```c
#ifndef WL_QUEUE_H
#define WL_QUEUE_H

#include <pthread.h>

// ─── 队列节点 ───
typedef struct wl_node {
    void *data;             // 持有的数据（AVFrame* / AVPacket* 等）
    struct wl_node *next;
} wl_node_t;

// ─── 队列 ───
typedef struct {
    wl_node_t *head;
    wl_node_t *tail;
    int count;
    int max_size;           // 容量上限，0 = 无限
    pthread_mutex_t mutex;
    pthread_cond_t cond_not_empty;
    pthread_cond_t cond_not_full;
    bool aborted;
    const char *name;       // 调试用
} wl_queue_t;

// 创建/销毁
wl_queue_t *wl_queue_create(int max_size, const char *name);
void wl_queue_destroy(wl_queue_t *q);

// 入队：队满则阻塞等待；abort 时丢弃返回 WL_ERR_EOF
int wl_queue_push(wl_queue_t *q, void *data);

// 出队：block=YES 空队列阻塞；block=NO 空队列返回 NULL
// 阻塞模式返回 NULL ⟺ 已 abort
void *wl_queue_pop(wl_queue_t *q, bool block);

// 带超时出队（毫秒）；超时返回 NULL
void *wl_queue_pop_timeout(wl_queue_t *q, int timeout_ms);

// 唤醒所有等待者并拒绝后续操作
void wl_queue_abort(wl_queue_t *q);

// 清空队列（不释放节点持有的 data，由调用方处理）
void wl_queue_flush(wl_queue_t *q);

int wl_queue_count(wl_queue_t *q);

#endif // WL_QUEUE_H
```

**实现要点**（`libwl/src/wl_queue.c`）：
- `pthread_mutex_t` 保护链表操作
- `pthread_cond_t` 做阻塞等待（not_empty / not_full）
- `wl_queue_pop` 返回 NULL 唯一语义 = 已 abort
- `wl_queue_flush` 遍历链表 free 节点（不 free data，data 归调用方管理）

## 第五步：C 内核 — wl_decoder.h（解码器）

你来写。这是 M1 的核心。

**文件**：`libwl/include/wl_decoder.h`

```c
#ifndef WL_DECODER_H
#define WL_DECODER_H

#include "wl_common.h"

typedef struct wl_decoder wl_decoder_t;

// ─── 回调函数类型 ───
// 解码器内部跑线程，通过回调把帧推给调用方
typedef void (*wl_video_output_cb)(wl_video_frame *frame, void *userdata);
typedef void (*wl_audio_output_cb)(wl_audio_frame *frame, void *userdata);

// ─── 创建/销毁 ───
wl_decoder_t *wl_decoder_create(const char *path);
void wl_decoder_destroy(wl_decoder_t **dec);

// ─── 信息查询 ───
double wl_decoder_duration(wl_decoder_t *dec);      // 总时长（秒）
int wl_decoder_width(wl_decoder_t *dec);             // 视频宽
int wl_decoder_height(wl_decoder_t *dec);            // 视频高
int wl_decoder_sample_rate(wl_decoder_t *dec);       // 音频采样率
int wl_decoder_channels(wl_decoder_t *dec);          // 音频声道数

// ─── 异步播放 ───
// 启动内部线程：parse → decode video → decode audio → 通过回调输出
// video_cb / audio_cb 可以为 NULL（表示不输出该类型）
void wl_decoder_start(wl_decoder_t *dec,
                      wl_video_output_cb video_cb,
                      wl_audio_output_cb audio_cb,
                      void *userdata);

void wl_decoder_stop(wl_decoder_t *dec);

// ─── 暂停/恢复 ───
void wl_decoder_set_paused(wl_decoder_t *dec, bool paused);
bool wl_decoder_is_paused(wl_decoder_t *dec);

// ─── 当前播放位置 ───
double wl_decoder_current_time(wl_decoder_t *dec);

#endif // WL_DECODER_H
```

### 5.1 wl_decoder 内部架构

参照 `WLMediaSource.m` 的 5 线程模型，但用纯 C + pthread 实现：

```
┌─────────────┐
│ parse thread │  av_read_frame → 按 stream_index 分发
└──────┬───────┘
       │
  ┌────┴────┐
  ▼         ▼
┌──────┐ ┌──────┐
│v_pkt │ │a_pkt │   线程安全队列（wl_queue_t）
│queue  │ │queue  │
└──┬───┘ └──┬───┘
   ▼        ▼
┌──────┐ ┌──────┐
│v_dec │ │a_dec │   decode threads：send_packet + receive_frame
│thread│ │thread│
└──┬───┘ └──┬───┘
   ▼        ▼
┌──────┐ ┌──────┐
│v_out │ │a_out │   回调输出给 OC 层
│      │ │      │
└──────┘ └──────┘
```

### 5.2 解码器内部数据结构

```c
// wl_decoder.c 内部
struct wl_decoder {
    AVFormatContext *fmt_ctx;
    AVCodecContext  *video_codec_ctx;
    AVCodecContext  *audio_codec_ctx;
    int video_stream_idx;
    int audio_stream_idx;
    double video_time_base;   // av_q2d(stream->time_base)
    double audio_time_base;

    wl_queue_t *video_pkt_queue;   // AVPacket* 队列
    wl_queue_t *audio_pkt_queue;

    // 线程
    pthread_t parse_thread;
    pthread_t video_dec_thread;
    pthread_t audio_dec_thread;

    // 回调
    wl_video_output_cb video_cb;
    wl_audio_output_cb audio_cb;
    void *userdata;

    // 状态
    volatile bool running;
    volatile bool paused;
    double current_time;      // 秒

    // VideoToolbox 硬解
    AVBufferRef *hw_device_ctx;

    // 软解转换
    struct SwsContext *sws_ctx;
    struct SwrContext *swr_ctx;
};
```

### 5.3 关键实现流程

#### parse_thread
```
1. while (running):
2.   检查 seek 请求（如有，执行 avformat_seek_file）
3.   av_read_frame(fmt_ctx, pkt)
4.   if EOF → 可选循环播放（seek 回起点）
5.   if pkt->stream_index == video_idx → 压拷贝入 video_pkt_queue
6.   if pkt->stream_index == audio_idx → 压拷贝入 audio_pkt_queue
7.   av_packet_unref(pkt)
8. 结束时 abort 两个 pkt_queue
```

#### video_dec_thread
```
1. while (running):
2.   从 video_pkt_queue pop（阻塞，30ms 超时）
3.   avcodec_send_packet + avcodec_receive_frame
4.   拿到 AVFrame → 转换为 BGRA CVPixelBuffer 或 uint8_t* buffer
5.   填充 wl_video_frame，调用 video_cb 回调
6.   free 帧数据由回调负责（或回调内部拷贝后立即 free）
```

#### audio_dec_thread
```
1. while (running):
2.   从 audio_pkt_queue pop（阻塞，30ms 超时）
3.   avcodec_send_packet + avcodec_receive_frame
4.   拿到 AVFrame → swr_convert 转为 interleaved float PCM
5.   填充 wl_audio_frame，调用 audio_cb 回调
```

### 5.4 VideoToolbox 硬解

参照 `WLMediaSource.m:887-895`：

```c
static int setup_hw_decoder(AVCodecContext *ctx) {
    AVBufferRef *hw_device_ctx = NULL;
    int err = av_hwdevice_ctx_create(&hw_device_ctx,
                                     AV_HWDEVICE_TYPE_VIDEOTOOLBOX,
                                     NULL, NULL, 0);
    if (err < 0) return err;
    ctx->hw_device_ctx = av_buffer_ref(hw_device_ctx);
    av_buffer_unref(&hw_device_ctx);
    return 0;
}
```

硬解帧直接从 `frame->data[3]` 取 `CVPixelBufferRef`（零拷贝）。

### 5.5 软解 → BGRA 转换

参照 `WLMediaSource.m:599-649`：

```c
// sws_scale 转为 BGRA
struct SwsContext *sws = sws_getContext(w, h, src_fmt,
                                        w, h, AV_PIX_FMT_BGRA,
                                        SWS_BILINEAR, NULL, NULL, NULL);
// 分配 CVPixelBuffer（带 IOSurface + Metal 兼容属性）
// 或者直接 malloc 一块 BGRA buffer
uint8_t *dst[1] = { bgre_buffer };
int dst_stride[1] = { width * 4 };
sws_scale(sws, frame->data, frame->linesize, 0, h, dst, dst_stride);
```

### 5.6 音频重采样

参照 `WLMediaSource.m:652-737`：

```c
// swr_convert 转为 interleaved float
swr_alloc_set_opts2(&swr_ctx,
                    &out_layout, AV_SAMPLE_FMT_FLT, src_rate,
                    &frame->ch_layout, src_fmt, src_rate,
                    0, NULL);
swr_init(swr_ctx);
// 分配 float buffer
float *out_buf = malloc(nb_samples * channels * sizeof(float));
uint8_t *out[1] = { (uint8_t*)out_buf };
swr_convert(swr_ctx, out, nb_samples,
            (const uint8_t**)frame->extended_data, nb_samples);
```

## 第六步：命令行测试 — wl_decoder_test.c

先在纯 C 环境验证解码器能跑，再接 OC。

**文件**：`libwl/tests/wl_decoder_test.c`

```c
#include <stdio.h>
#include "wl_decoder.h"

static int video_count = 0;
static int audio_count = 0;

void on_video(wl_video_frame *frame, void *ud) {
    video_count++;
    printf("[VIDEO] #%d  pts=%.3f  %dx%d  linesize=%d\n",
           video_count, frame->pts, frame->width, frame->height, frame->linesize);
    // 只打印前 5 帧
    if (video_count >= 5) {
        wl_decoder_t *dec = (wl_decoder_t *)ud;
        wl_decoder_stop(dec);
    }
    wl_video_frame_free(frame);
}

void on_audio(wl_audio_frame *frame, void *ud) {
    audio_count++;
    if (audio_count <= 5) {
        printf("[AUDIO] #%d  pts=%.3f  samples=%d  ch=%d  rate=%d\n",
               audio_count, frame->pts, frame->nb_samples,
               frame->channels, frame->sample_rate);
    }
    wl_audio_frame_free(frame);
}

int main(int argc, char *argv[]) {
    const char *path = argc > 1 ? argv[1] : "test.mp4";
    printf("Opening: %s\n", path);

    wl_decoder_t *dec = wl_decoder_create(path);
    if (!dec) { fprintf(stderr, "create failed\n"); return 1; }

    printf("Duration: %.1fs  Video: %dx%d  Audio: %dHz %dch\n",
           wl_decoder_duration(dec),
           wl_decoder_width(dec), wl_decoder_height(dec),
           wl_decoder_sample_rate(dec), wl_decoder_channels(dec));

    wl_decoder_start(dec, on_video, on_audio, dec);

    // 等待播放结束（简单 sleep，实际用 pthread_join 或条件变量）
    sleep(3);

    wl_decoder_destroy(&dec);
    printf("Done. video=%d audio=%d\n", video_count, audio_count);
    return 0;
}
```

## 第九步：OC 胶水 — WLPlayer（播放器协调器）

我来写。这是胶水层的核心，桥接 C 解码器和 OC 渲染/播放。

**职责**：
1. 创建 `wl_decoder_t`
2. 注册回调：video_cb → `CVPixelBuffer` → `WLMetalRenderer`
3. 注册回调：audio_cb → `WLAudioPlayer`
4. 管理播放状态（播放/暂停/停止）

**桥接关键点**：
- video_cb 在 C 线程调用，需要 `dispatch_async` 到主线程/Metal 线程渲染
- audio_cb 在 C 线程调用，直接送 AudioQueue（AudioQueue 内部线程安全）
- `wl_video_frame.data` 是 `uint8_t*` BGRA → 包装成 `CVPixelBufferRef` → Metal 纹理
