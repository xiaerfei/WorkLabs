# 源注册机制设计

> 创建日期：2026-06-30
> 对标：OBS `obs_source_info` + `obs_register_source()`（`libobs/obs-source.c`）
> 状态：设计阶段

---

## 一、为什么需要注册机制

WorkOBS 需要支持多种源类型：

| 源类型 | 帧从哪来 | 驱动方式 | 是否需要解码 |
|--------|---------|---------|-------------|
| 本地媒体文件 | FFmpeg 解码 → CVPixelBuffer / PCM | `wl_media_thread` 主动拉文件 | 是 |
| Camera | 硬件直接给 CVPixelBuffer | AVFoundation 回调被动通知 | 否 |
| 麦克风 | 硬件直接给 PCM | CoreAudio 回调被动通知 | 否 |
| 网络流（RTMP/RTSP/HLS） | 网络拉流 + FFmpeg 解码 | 类似本地文件 | 是 |
| 屏幕采集 | CGDisplayStream / ScreenCaptureKit | 系统回调 | 否 |

它们的**生产方式**完全不同，但**消费方式**完全相同——都走 `output_video_frame` → `async_frames` 和 `output_audio_frame` → `audio_input_buf`，下游（全局 render 线程）不感知来源。

OBS 的设计哲学（`OBS_媒体源线程模型.md` §六）：

> **不管源是什么，统一走 `obs_source_output_video` / `obs_source_output_audio` 入队，下游完全不感知来源。**

注册机制就是实现这个哲学的基础设施：定义一套统一接口（vtable），每种源类型实现自己的 vtable，运行时注册到全局表，按 type name 查找创建实例。

### 2.2 obs_register_source

```c
// OBS libobs/obs-source.c（简化）
void obs_register_source(struct obs_source_info *info) {
    // 加入全局链表
    da_push_back(obs->source_types, &info);
}

// 按 id 查找
static obs_source_info *get_source_info(const char *id) {
    for each info in obs->source_types:
        if (strcmp(info->id, id) == 0) return info;
    return NULL;

```

### 2.3 obs_source（实例）

```c
// 创建源实例
obs_source_t *obs_source_create(const char *id, const char *name, ...) {
    obs_source_info *info = get_source_info(id);  // 查注册表
    void *data = info->create(settings, source);   // 调 vtable.create
    // ...
}
```

**关键分离**：`obs_source_info` 是**类型描述**（全局唯一，静态），`obs_source` 是**运行时实例**（可有多个）。

## 三、wl_source_info 设计（vtable）

### 3.1 结构体定义

```c
// wl_source_info.h

typedef struct wl_source_info wl_source_info_t;

/**
 * 源类型描述（vtable）。
 * 每种源类型定义一个 wl_source_info 实例，注册到全局表。
 * 对齐 OBS obs_source_info。
 */
struct wl_source_info {
    // ─── 标识 ───
    const char *id;          // 类型唯一标识，如 "media_file", "camera", "mic"
    const char *type_name;   // 显示名，如 "Media File", "Camera", "Microphone"

    // ─── 生命周期 ───

    /**
     * 创建源实例（分配私有数据，初始化资源）。
     * @param settings  配置参数（文件路径 / 设备名 / URL 等）
     * @return 私有数据指针（传给后续所有回调的 data 参数）
     */
    void *(*create)(const char *settings);

    /**
     * 销毁源实例（释放所有资源）。
     * 调用前必须已 stop。
     */
    void  (*destroy)(void *data);

    // ─── 控制 ───

    /**
     * 启动源（开始产出帧）。
     * - 媒体文件：启动 wl_media_thread
     * - Camera：启动 AVCaptureSession
     * - 麦克风：启动 AudioUnit
     */
    int   (*start)(void *data);

    /**
     * 停止源（停止产出帧，join 内部线程）。
     */
    void  (*stop)(void *data);

    /**
     * 暂停 / 恢复（可选，不需要暂停能力的源可为 NULL）。
     */
    void  (*pause)(void *data, bool paused);

    /**
     * Seek 到指定位置（可选，仅媒体文件类源需要）。
     * @param seek_ts_us  目标时间戳（微秒）
     */
    void  (*seek)(void *data, int64_t seek_ts_us);

    // ─── 信息查询（可选，NULL 表示不支持）───

    /**
     * 获取源的时长（微秒）。仅媒体文件类源有意义。
     */
    int64_t (*get_duration)(void *data);

    /**
     * 获取源的视频尺寸。Camera / 媒体文件有意义，纯音频源返回 0。
     */
    void    (*get_video_size)(void *data, int *width, int *height);
};
```

### 3.2 回调职责划分

| 回调 | 媒体文件源 | Camera 源 | 麦克风源 |
|------|-----------|----------|---------|
| `create` | 创建 decoder + media_thread | 创建 AVCaptureSession | 创建 AudioUnit |
| `destroy` | 释放 decoder + thread | 停止 session + 释放 | 停止 unit + 释放 |
| `start` | `wl_media_thread_start()` | `[session startRunning]` | `AudioOutputUnitStart()` |
| `stop` | `wl_media_thread_stop()` | `[session stopRunning]` | `AudioOutputUnitStop()` |
| `pause` | `wl_media_thread_pause()` | NULL（不支持） | NULL |
| `seek` | `wl_media_thread_seek()` | NULL | NULL |
| `get_duration` | 返回文件总时长 | NULL | NULL |
| `get_video_size` | 返回视频分辨率 | 返回采集分辨率 | 返回 0, 0 |

## 五、注册表设计

### 5.1 数据结构

```c
// wl_source_registry.c（内部）

#define MAX_SOURCE_TYPES 64

static struct {
    wl_source_info_t *entries[MAX_SOURCE_TYPES];
    int               count;
    pthread_mutex_t   mutex;
} g_registry = {
    .count = 0,
    .mutex = PTHREAD_MUTEX_INITIALIZER,
};
```

用固定数组而非动态数组，因为源类型数量极少（< 20），编译期确定上限，无 malloc 开销。

### 5.2 注册 / 注销

```c
// wl_source_registry.h

/**
 * 注册一种源类型。通常在模块初始化时调用。
 * 线程安全（加锁写入）。
 * 重复注册同 id → 覆盖（打 warning）。
 */
void wl_source_register(wl_source_info_t *info);

/**
 * 注销一种源类型。通常在模块卸载时调用。
 */
void wl_source_unregister(const char *id);
```

### 5.3 查找

```c
/**
 * 按 id 查找已注册的源类型。
 * 返回 NULL 表示未找到。
 * 线程安全（加锁读取）。
 */
wl_source_info_t *wl_source_find(const char *id);
```

### 5.4 实现

```c
void wl_source_register(wl_source_info_t *info) {
    pthread_mutex_lock(&g_registry.mutex);

    // 检查重复
    for (int i = 0; i < g_registry.count; i++) {
        if (strcmp(g_registry.entries[i]->id, info->id) == 0) {
            fprintf(stderr, "[registry] warning: source '%s' already registered, overwriting\n",
                    info->id);
            g_registry.entries[i] = info;
            pthread_mutex_unlock(&g_registry.mutex);
            return;
        }
    }

    // 新注册
    if (g_registry.count < MAX_SOURCE_TYPES) {
        g_registry.entries[g_registry.count++] = info;
    } else {
        fprintf(stderr, "[registry] error: source type table full\n");
    }

    pthread_mutex_unlock(&g_registry.mutex);
}

wl_source_info_t *wl_source_find(const char *id) {
    pthread_mutex_lock(&g_registry.mutex);
    for (int i = 0; i < g_registry.count; i++) {
        if (strcmp(g_registry.entries[i]->id, id) == 0) {
            wl_source_info_t *found = g_registry.entries[i];
            pthread_mutex_unlock(&g_registry.mutex);
            return found;
        }
    }
    pthread_mutex_unlock(&g_registry.mutex);
    return NULL;
}
```

## 七、wl_source_create 实现

```c
wl_source_t *wl_source_create(const char *type_id, const char *settings) {
    // 1. 查注册表
    wl_source_info_t *info = wl_source_find(type_id);
    if (!info) {
        fprintf(stderr, "[source] unknown type '%s'\n", type_id);
        return NULL;
    }

    // 2. 分配实例
    wl_source_t *src = calloc(1, sizeof(*src));
    if (!src) return NULL;

    src->info = info;

    // 3. 调 vtable.create 创建私有数据
    src->data = info->create(settings);
    if (!src->data) {
        fprintf(stderr, "[source] create failed for type '%s'\n", type_id);
        free(src);
        return NULL;
    }

    // 4. 初始化输出缓冲（M3/M4 实现）
    // pthread_mutex_init(&src->async_mutex, NULL);
    // src->async_num = 0;

    return src;
}
```

## 九、源类型全对比

| | 本地媒体文件 | Camera | 麦克风 |
|--|------------|--------|--------|
| vtable id | `"media_file"` | `"camera"` | `"mic"` |
| 帧驱动方式 | `wl_media_thread` 主动拉 | AVFoundation 回调被动推 | CoreAudio 回调被动推 |
| 视频输出 | `wl_source_output_video()` | `wl_source_output_video()` | 无 |
| 音频输出 | `wl_source_output_audio()` | 无 | `wl_source_output_audio()` |
| 解码 | FFmpeg（VideoToolbox / 软解） | 无需解码 | 无需解码 |
| pause | ✅ | ❌ | ❌ |
| seek | ✅ | ❌ | ❌ |
| get_duration | ✅ | ❌ | ❌ |

**关键**：不管驱动方式如何，所有源都调用同一个 `wl_source_output_video` / `wl_source_output_audio` 推帧到缓冲区，下游完全不感知来源。

## 十一、初始化时序

```c
// 应用启动时（main / AppDelegate）

// 1. 注册所有源类型
wl_source_register(&g_media_source_info);
wl_source_register(&g_camera_source_info);
wl_source_register(&g_mic_source_info);

// 2. 用户添加源（如打开一个文件）
wl_source_t *src = wl_source_create("media_file", "/path/to/video.mp4");

// 3. 启动源
wl_source_start(src);

// 4. 源内部开始产出帧（media_thread 调用 output_video / output_audio）

// 5. 全局 render 线程从 async_frames / audio_buf 消费帧

// 6. 用户暂停
wl_source_pause(src, true);

// 7. 用户 seek
wl_source_seek(src, 30 * 1000000);  // 30秒

// 8. 用户移除源
wl_source_destroy(src);  // 内部先 stop 再 destroy
```
